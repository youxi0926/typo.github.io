Typo Generator

ドメインを入力すると、タイポ候補とスコアを自動生成します。



  
from js import document
from pyodide.ffi import create_proxy
from collections import defaultdict

# 簡易キーボード隣接キー定義
keyboard_adjacent = {
    'a': 'qwsz',
    's': 'awedxz',
    'd': 'serfcx',
    'f': 'drtgcv',
    'g': 'ftyhbv',
    'h': 'gyujnb',
    'j': 'huikmn',
    'k': 'jiolm,',
    'l': 'kop;.',
}

def simple_score(base, typo):
    """単純なスコア関数（差分が少ないほど高スコア）"""
    diff = sum(1 for a, b in zip(base, typo) if a != b)
    return round(1 / (1 + diff), 3)

def generate_typos(domain):
    """ドメイン名からタイポ候補を生成"""
    variants = defaultdict(set)

    for i in range(len(domain)):
        c = domain[i]

        # 入力漏れ
        typo = domain[:i] + domain[i+1:]
        variants[typo].add("入力漏れ")

        # 二重入力
        typo = domain[:i] + c + c + domain[i+1:]
        variants[typo].add("二重入力")

        # 隣接キー誤打
        for adj in keyboard_adjacent.get(c.lower(), ''):
            typo = domain[:i] + adj + domain[i+1:]
            variants[typo].add("隣接キー誤打")

    # ドット抜け
    if '.' in domain:
        typo = domain.replace('.', '', 1)
        variants[typo].add("ドット抜け")

    # 入力順序ミス
    for i in range(len(domain) - 1):
        swapped = domain[:i] + domain[i+1] + domain[i] + domain[i+2:]
        variants[swapped].add("入力順序ミス")

    return variants

def run(event):
    domain = document.getElementById("domain").value.strip()
    output_elem = document.getElementById("output")

    if not domain:
        output_elem.innerText = "⚠️ ドメインを入力してください。"
        return

    variants = generate_typos(domain)
    results = []

    for typo, causes in variants.items():
        if typo == domain:
            continue
        score = simple_score(domain, typo)
        results.append((typo, "・".join(causes), score))

    if not results:
        output_elem.innerText = "タイポ候補が生成されませんでした。"
        return

    results.sort(key=lambda x: x[2], reverse=True)

    text = f"『{domain}』のタイポ候補（上位10件）:\n\n"
    for i, (typo, cause, score) in enumerate(results[:10], 1):
        text += f"{i:>2}位: {typo:<25} | 原因: {cause:<12} | スコア: {score}\n"

    output_elem.innerText = text

document.getElementById("run-btn").addEventListener("click", create_proxy(run))
